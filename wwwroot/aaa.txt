The error ReferenceError: sortDataset is not defined means the helper function was either not added or is placed in a different scope where dataSrc cannot reach it.
You have two simple options to fix this:
Option 1: Place the helper function right above the table (Recommended)
Place this function directly before line const tableSBOMNew = new DataTable(...) in newSite.js:
function sortDataset(dataArray) {
    return dataArray.sort(function (a, b) {
        var rankA = (typeof customOrder !== 'undefined' && customOrder[a.assetCategory]) ? customOrder[a.assetCategory] : 999;
        var rankB = (typeof customOrder !== 'undefined' && customOrder[b.assetCategory]) ? customOrder[b.assetCategory] : 999;
        if (rankA !== rankB) return rankA - rankB;

        if (a.assetId !== b.assetId) return (a.assetId || 0) - (b.assetId || 0);
        if (a.componentId !== b.componentId) return (a.componentId || 0) - (b.componentId || 0);
        return (a.cryptoId || 0) - (b.cryptoId || 0);
    });
}

Option 2: Put the sorting inline directly in dataSrc
If you don't want a separate helper function, replace the line list = sortDataset(list); with this direct inline sort:
dataSrc: function (json) {
    var list = Array.isArray(json) ? json : (json.data || []);

    // Sort inline without calling an external function
    list.sort(function (a, b) {
        var rankA = (typeof customOrder !== 'undefined' && customOrder[a.assetCategory]) ? customOrder[a.assetCategory] : 999;
        var rankB = (typeof customOrder !== 'undefined' && customOrder[b.assetCategory]) ? customOrder[b.assetCategory] : 999;
        if (rankA !== rankB) return rankA - rankB;

        if (a.assetId !== b.assetId) return (a.assetId || 0) - (b.assetId || 0);
        if (a.componentId !== b.componentId) return (a.componentId || 0) - (b.componentId || 0);
        return (a.cryptoId || 0) - (b.cryptoId || 0);
    });

    var assetCounter = 0;
    var lastAssetId = null;
    var compCounter = 0;

    for (var i = 0; i < list.length; i++) {
        var item = list[i];

        if (item.assetId !== lastAssetId) {
            assetCounter++;
            lastAssetId = item.assetId;
            compCounter = 0;
        }

        compCounter++;
        item.displayIndex = assetCounter + "." + compCounter;
    }

    return list;
}
