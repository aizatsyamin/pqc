$(document).on('change keyup', '.inputCryptoCertValidityNumber, .selectCryptoCertValidity', function () {
    var gNum = parseInt($('.inputCryptoCertValidityNumber').val(), 10) || 0;
    var gType = ($('.selectCryptoCertValidity').val() || '').toLowerCase();

    // Start with today's date
    var targetDate = new Date();

    if (gType.includes('day')) {
        targetDate.setDate(targetDate.getDate() + gNum);
    } else if (gType.includes('month')) {
        targetDate.setMonth(targetDate.getMonth() + gNum);
    } else if (gType.includes('year')) {
        targetDate.setFullYear(targetDate.getFullYear() + gNum);
    }

    // Check if target year exceeds 2030
    var isPast2030 = targetDate.getFullYear() > 2030;

    if (isPast2030) {
        // Exceeds year 2030
        console.warn("Target date exceeds 2030:", targetDate);
        Swal.fire({
            icon: 'warning',
            title: 'Invalid Validity',
            text: 'Validity cannot extend past the year 2030.',
            confirmButtonColor: '#1C2534'
        });
    }
});
