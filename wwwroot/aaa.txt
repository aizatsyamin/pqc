To make the index hierarchy clean (1, 1.1, 1.1.1) instead of showing raw database primary keys (like 4.4 or 4.4.3), generate a shared mapping index based on the ordered Inventory rows, and then use that map in SBOM and CBOM.
Step 1: Create Global Map Dictionaries
Place this above your DataTable initializations in newSite.js:
// Global maps to keep numbering consistent across tables
var inventoryMap = {}; // assetId -> 1, 2, 3...
var sbomMap = {};      // assetId_componentId -> 1.1, 1.2...
var cbomCounter = {};  // assetId_componentId -> count of cryptos (1, 2...)

Step 2: Inventory Table (Builds 1, 2, 3, 4)
Inside your tableInventoryNew configuration:
 * In the first column definition (columns[0]):
{
  "data": function (row, type, set, meta) {
    if (type === 'display' || type === 'filter') {
      return inventoryMap[row.assetId] || (meta.row + 1);
    }
    return row.assetId;
  }
}

 * Inside ajax.dataFilter or initComplete / drawCallback to populate inventoryMap based on display order:
drawCallback: function (settings) {
  var api = this.api();
  // Clear map on draw
  inventoryMap = {};
  
  // Assign 1, 2, 3... based on current table sort/order
  api.rows({ search: 'applied', order: 'applied' }).every(function (rowIdx, tableLoop, rowLoop) {
    var data = this.data();
    if (data && data.assetId !== undefined) {
      inventoryMap[data.assetId] = rowLoop + 1;
    }
  });

  // Redraw the first column text cleanly
  api.rows().invalidate('display');
}

Step 3: SBOM Table (Generates 1.1, 1.2, 2.1...)
In tableSBOMReport, use inventoryMap for the prefix and maintain an internal count for each component:
 * Add a reset in preDrawCallback:
preDrawCallback: function (settings) {
  sbomMap = {};
  var compCounts = {};

  var api = this.api();
  api.rows({ search: 'applied', order: 'applied' }).every(function () {
    var d = this.data();
    var aId = d.assetId;
    var cId = d.componentId;
    var key = aId + '_' + cId;

    if (!sbomMap[key]) {
      compCounts[aId] = (compCounts[aId] || 0) + 1;
      var invIndex = inventoryMap[aId] || aId; // Uses 1, 2, etc. from Inventory
      sbomMap[key] = invIndex + '.' + compCounts[aId]; // e.g. 1.1, 1.2
    }
  });
}

 * In the SBOM first column definition:
{
  "data": function (row, type) {
    var key = row.assetId + '_' + row.componentId;
    return sbomMap[key] || (row.assetId + '.' + row.componentId);
  }
}

{
  "data": function (row, type) {
    var sbomKey = row.assetId + '_' + row.componentId;
    var sbomPrefix = sbomMap[sbomKey] || (row.assetId + '.' + row.componentId);

    // Track running count of cryptos for this specific asset+component
    if (type === 'display' || type === 'filter') {
      var cryptoTrackingKey = row.assetId + '_' + row.componentId + '_' + row.cryptoId;
      if (!this._cbomCache) this._cbomCache = {};

      if (!this._cbomCache[cryptoTrackingKey]) {
        cbomCounter[sbomKey] = (cbomCounter[sbomKey] || 0) + 1;
        this._cbomCache[cryptoTrackingKey] = sbomPrefix + '.' + cbomCounter[sbomKey];
      }
      return this._cbomCache[cryptoTrackingKey];
    }

    return row.assetId + '.' + row.componentId + '.' + row.cryptoId;
  }
}
