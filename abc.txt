-- 1. Create the sequence
CREATE SEQUENCE SEQ_FE_STKID
  START WITH 1
  INCREMENT BY 1
  NOCACHE
  NOCYCLE;

-- 2. Create the auto-increment trigger
CREATE OR REPLACE TRIGGER TRG_FE_INVSTOCK_BI
BEFORE INSERT ON FE_INVSTOCK
FOR EACH ROW
BEGIN
    -- If FE_STKID is not provided, generate the next value from the sequence
    IF :NEW.FE_STKID IS NULL OR :NEW.FE_STKID = 0 THEN
        :NEW.FE_STKID := SEQ_FE_STKID.NEXTVAL;
    END IF;
END;
/
