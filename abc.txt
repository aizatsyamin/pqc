$('.moneyFormat').text((_, currentText) => {
  const num = parseFloat(currentText.replace(/[^0-9.-]+/g, '')) || 0;
  return moneyFormatter.format(num);
});

$('.numberFormat').text((_, currentText) => {
  const num = parseFloat(currentText.replace(/[^0-9.-]+/g, '')) || 0;
  return num.toLocaleString();
});
